2.3 ADVANCED RAG ENGINE (Multi-Index Hybrid)
Pattern: Retrieve from multiple indices simultaneously, rerank, compress.
Python

class HybridRAG:
    def __init__(self):
        self.vector_store = PineconeVectorStore()
        self.sparse_retriever = BM25Retriever()
        self.graph_store = Neo4jGraph()
        self.reranker = CrossEncoder("cross-encoder/ms-marco-MiniLM-L-6-v2")
        
    async def retrieve(self, query: str, context: dict) -> RetrievalResult:
        # Parallel retrieval
        vector_task = self.vector_store.asimilarity_search(query, k=10)
        sparse_task = self.sparse_retriever.aretrieve(query, k=10)
        graph_task = self._graph_traversal(query)
        
        vector_docs, sparse_docs, graph_context = await asyncio.gather(
            vector_task, sparse_task, graph_task
        )
        
        # Fusion & Reranking
        fused = self._reciprocal_rank_fusion(vector_docs, sparse_docs)
        reranked = self.reranker.rerank(query, fused, top_k=5)
        
        # Context compression (token-aware)
        compressed = self._compress_context(reranked, max_tokens=4000)
        
        return RetrievalResult(
            documents=compressed,
            graph_context=graph_context,
            sources=self._extract_sources(compressed)
        )
    
    async def _graph_traversal(self, query: str) -> str:
        """
        Extract entities from query, find in graph, expand neighborhood.
        """
        entities = await self._extract_entities(query)
        subgraph = await self.graph_store.query(
            f"""
            MATCH (e:Entity)-[r]-(related)
            WHERE e.name IN {entities}
            RETURN e, r, related
            LIMIT 20
            """
        )
        return self._format_graph_context(subgraph)
Retrieval Flow:
Query Rewriting: Expand query with synonyms, handle typos
Parallel Retrieval: Vector (semantic) + Sparse (lexical) + Graph (relational)
Reciprocal Rank Fusion: Combine results without score calibration
Cross-Encoder Reranking: Precise relevance scoring
Context Compression: Remove redundancy, preserve key facts
Citation Mapping: Track sources for grounding

class SafeSQLExecutor:
    def __init__(self):
        self.allowed_schemas = ["analytics", "reporting"]  # Read-only
        self.query_validator = QueryValidator()
        self.connection_pool = AsyncConnectionPool(
            dsn="postgresql://read-replica/analytics",
            max_size=20
        )
        
    async def execute(self, nl_query: str, context: dict) -> SQLResult:
        # Step 1: Schema introspection
        schema = await self._get_relevant_schema(nl_query)
        
        # Step 2: SQL Generation with few-shot examples
        sql = await self._generate_sql(nl_query, schema, context)
        
        # Step 3: Validation (multi-layer)
        if not self._is_safe(sql):
            raise SecurityError("Unsafe SQL detected")
            
        if not self._is_select_only(sql):
            raise SecurityError("Only SELECT queries allowed")
            
        # Step 4: Execution on read replica with timeout
        try:
            async with self.connection_pool.acquire() as conn:
                result = await asyncio.wait_for(
                    conn.fetch(sql), 
                    timeout=10.0
                )
        except asyncio.TimeoutError:
            return SQLResult(error="Query timeout - too complex")
            
        # Step 5: Result interpretation
        summary = await self._summarize_results(result, nl_query)
        
        return SQLResult(
            sql=sql,
            data=result,
            summary=summary,
            row_count=len(result)
        )
        
    def _is_safe(self, sql: str) -> bool:
        """Check for SQL injection patterns"""
        forbidden = [";", "--", "drop", "delete", "update", "insert"]
        return not any(f in sql.lower() for f in forbidden)

class MemoryManager:
    def __init__(self):
        # Layer 1: Working Memory (Redis/In-Memory)
        self.working_memory = RedisCache(ttl=3600)
        
        # Layer 2: Episodic Memory (Vector DB + Time-series)
        self.episodic_store = Mem0Client()  # or custom implementation
        
        # Layer 3: Semantic Memory (Graph DB)
        self.semantic_graph = Neo4jGraph()
        
    async def process_interaction(self, user_msg: str, assistant_msg: str, 
                                  user_id: str, session_id: str):
        # 1. Update Working Memory (immediate context)
        await self._update_working_memory(session_id, user_msg, assistant_msg)
        
        # 2. Extract salient facts for Episodic Memory
        facts = await self._extract_facts(user_msg, assistant_msg)
        
        # 3. Store in Episodic with embeddings
        for fact in facts:
            await self.episodic_store.add(
                content=fact,
                user_id=user_id,
                session_id=session_id,
                timestamp=datetime.utcnow(),
                metadata={"type": "conversation_fact"}
            )
            
        # 4. Update Semantic Memory (entities & relations)
        await self._update_knowledge_graph(user_id, facts)
        
    async def retrieve_context(self, query: str, user_id: str, 
                              session_id: str) -> MemoryContext:
        # Retrieve from all layers
        working = await self.working_memory.get(session_id)
        episodic = await self.episodic_store.search(
            query=query,
            user_id=user_id,
            limit=10
        )
        semantic = await self._query_graph(user_id, query)
        
        return MemoryContext(
            working_memory=working,
            relevant_episodes=episodic,
            semantic_context=semantic
        )

class ToolRegistry:
    def __init__(self):
        self.tools: dict[str, BaseTool] = {}
        self.schemas: dict[str, dict] = {}
        
    def register(self, tool: BaseTool):
        self.tools[tool.name] = tool
        self.schemas[tool.name] = tool.get_schema()
        
    async def execute(self, tool_call: ToolCall) -> ToolResult:
        tool = self.tools.get(tool_call.name)
        if not tool:
            return ToolResult(error=f"Tool {tool_call.name} not found")
            
        # Pre-execution validation
        if not self._validate_params(tool_call):
            return ToolResult(error="Invalid parameters")
            
        # Execute with timeout and retry
        try:
            result = await asyncio.wait_for(
                tool.ainvoke(tool_call.arguments),
                timeout=tool.timeout
            )
            return ToolResult(data=result, success=True)
        except Exception as e:
            return ToolResult(error=str(e), success=False)

class StreamingPipeline:
    def __init__(self):
        self.kafka_producer = KafkaProducer()
        self.sse_manager = SSEManager()
        
    async def handle_query(self, query: str, user_id: str, session_id: str):
        # Initialize SSE stream
        stream_id = generate_uuid()
        await self.sse_manager.connect(session_id, stream_id)
        
        try:
            # Start pipeline
            async for event in self._process_stream(query, user_id, session_id):
                await self.sse_manager.send(session_id, event)
                
                # Log to Kafka for analytics
                self.kafka_producer.send("assistant-events", event)
                
        except Exception as e:
            await self.sse_manager.send(session_id, {
                "type": "error",
                "message": str(e)
            })
        finally:
            await self.sse_manager.disconnect(session_id)
            
    async def _process_stream(self, query, user_id, session_id):
        # Yield status updates
        yield {"type": "status", "message": "Analyzing intent..."}
        
        # Intent classification
        intent = await self.classifier.classify(query)
        yield {"type": "intent", "data": intent}
        
        # Memory retrieval
        yield {"type": "status", "message": "Retrieving context..."}
        memory = await self.memory.retrieve(query, user_id)
        
        # RAG Retrieval
        yield {"type": "status", "message": "Searching knowledge base..."}
        docs = await self.rag.retrieve(query)
        yield {"type": "sources", "data": [d.metadata for d in docs]}
        
        # Generation (streaming tokens)
        yield {"type": "status", "message": "Generating response..."}
        async for token in self.llm.astream(query, context=docs):
            yield {"type": "token", "data": token}
            
        yield {"type": "complete"}

class GuardrailSystem:
    def __init__(self):
        self.pii_detector = PresidioAnalyzer()
        self.content_filter = ContentModeration()
        self.grounding_checker = GroundingCheck()
        
    async def check_input(self, text: str) -> GuardResult:
        # PII Detection
        pii_results = self.pii_detector.analyze(text)
        if pii_results:
            return GuardResult(
                allowed=False,
                action="redact",
                redacted_text=self._redact_pii(text, pii_results)
            )
            
        # Toxicity check
        if self.content_filter.is_toxic(text):
            return GuardResult(allowed=False, action="block")
            
        return GuardResult(allowed=True)
        
    async def check_output(self, response: str, sources: list) -> GuardResult:
        # Hallucination check
        grounded = await self.grounding_checker.verify(response, sources)
        if not grounded.is_grounded:
            return GuardResult(
                allowed=False,
                action="retry",
                confidence=grounded.confidence
            )
            
        return GuardResult(allowed=True)
